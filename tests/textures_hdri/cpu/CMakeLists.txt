cmake_minimum_required(VERSION 3.20)
project(HoudiniRTTextureCpuTests LANGUAGES CXX)
find_package(Houdini REQUIRED)
enable_testing()
set(HRT_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../src/hdRT")
set(HRT_CPU "${HRT_SOURCE}/material.cpp" "${HRT_SOURCE}/materialTexture.cpp"
    "${HRT_SOURCE}/mesh.cpp" "${HRT_SOURCE}/meshNormals.cpp" "${HRT_SOURCE}/meshUV.cpp"
    "${HRT_SOURCE}/world.cpp" "${HRT_SOURCE}/light.cpp")
add_library(hrt_cpu STATIC ${HRT_CPU})
target_include_directories(hrt_cpu PUBLIC "${HRT_SOURCE}")
target_link_libraries(hrt_cpu PUBLIC Houdini HoudiniThirdParty)
target_compile_features(hrt_cpu PUBLIC cxx_std_20)
target_compile_definitions(hrt_cpu PUBLIC _SILENCE_CXX20_OLD_SHARED_PTR_ATOMIC_SUPPORT_DEPRECATION_WARNING)
if(MSVC)
    target_compile_options(hrt_cpu PRIVATE /MP4)
endif()
add_executable(texture_ingress texture_ingress.cpp)
add_executable(material_regression ../../reliable_materials/material_translation.cpp)
add_executable(normal_regression ../../reliable_lighting/mesh_normals.cpp)
foreach(target texture_ingress material_regression normal_regression)
    target_include_directories(${target} PRIVATE "${HRT_SOURCE}")
    target_link_libraries(${target} PRIVATE hrt_cpu)
    target_compile_features(${target} PRIVATE cxx_std_20)
    target_compile_definitions(${target} PRIVATE _SILENCE_CXX20_OLD_SHARED_PTR_ATOMIC_SUPPORT_DEPRECATION_WARNING)
    add_test(NAME ${target} COMMAND $<TARGET_FILE:${target}> "${CMAKE_CURRENT_LIST_DIR}/../../../build/m7/native-texture-fixture")
endforeach()
