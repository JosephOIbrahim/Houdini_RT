cmake_minimum_required(VERSION 3.20)
project(HoudiniRTAssetTests LANGUAGES CXX)
find_package(Houdini REQUIRED)
get_filename_component(HRT_HDK_ROOT "${Houdini_DIR}/../.." ABSOLUTE)
set(HRT_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../../../src/hdRT")
add_executable(asset_probe asset_probe.cpp "${HRT_SOURCE}/imageAssets.cpp")
target_include_directories(asset_probe PRIVATE "${HRT_SOURCE}")
target_compile_features(asset_probe PRIVATE cxx_std_20)
find_library(HRT_OIIO OpenImageIO_sidefx HINTS "${HRT_HDK_ROOT}/custom/houdini/dsolib" REQUIRED)
find_library(HRT_OIIO_UTIL OpenImageIO_Util_sidefx HINTS "${HRT_HDK_ROOT}/custom/houdini/dsolib" REQUIRED)
find_library(HRT_OCIO OpenColorIO_sidefx HINTS "${HRT_HDK_ROOT}/custom/houdini/dsolib" REQUIRED)
target_link_libraries(asset_probe PRIVATE Houdini HoudiniThirdParty ${HRT_OIIO} ${HRT_OIIO_UTIL} ${HRT_OCIO})
target_compile_definitions(asset_probe PRIVATE _SILENCE_CXX20_OLD_SHARED_PTR_ATOMIC_SUPPORT_DEPRECATION_WARNING)
